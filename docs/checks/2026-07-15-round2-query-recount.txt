# Mechanical recount of preregistered query strings in round-2 protocol v2.
# Re-runnable check: python recount_queries.py <path-to-protocol.md>
import re, sys

path = sys.argv[1]
lines = open(path, encoding="utf-8").read().splitlines()

in_yaml = False
lane = None
mode = None  # 'pq' | 'opt' | None
lanes = {}  # lane_id -> {'pq': [...], 'opt': [...], 'engines': str}

for ln in lines:
    if ln.strip().startswith("```yaml"):
        in_yaml = True; continue
    if ln.strip() == "```":
        in_yaml = False; lane = None; mode = None; continue
    if not in_yaml:
        continue
    m = re.match(r"^- lane_id:\s*(\S+)", ln)
    if m:
        lane = m.group(1)
        lanes[lane] = {"pq": [], "opt": [], "engines": ""}
        mode = None
        continue
    if lane is None:
        continue
    if re.match(r"^  engines:", ln):
        lanes[lane]["engines"] = ln
        mode = None
        continue
    if re.match(r"^  planned_queries:", ln):
        mode = "pq"; continue
    if re.match(r"^  optional_native_language_probes:", ln):
        mode = "opt"; continue
    if re.match(r"^  \S", ln):  # any other 2-space key ends the list
        mode = None
        continue
    if mode and re.match(r"^    - ", ln):
        lanes[lane][mode].append(ln.strip()[2:])

mand = sum(len(v["pq"]) for v in lanes.values())
opt = sum(len(v["opt"]) for v in lanes.values())
chase = sum(1 for v in lanes.values() for q in v["pq"]
            if "CITATION_BACKWARD" in q or "CITATION_FORWARD" in q)
site = sum(1 for v in lanes.values() for q in v["pq"] if "site:ieeexplore" in q)
cat = sum(1 for v in lanes.values() for q in v["pq"] if "cat:eess.AS" in q)

print(f"lanes_total: {len(lanes)}")
for k, v in lanes.items():
    tag = ""
    if "arXiv_listing" in v["engines"] and v["pq"]:
        has_cat = any("cat:eess.AS" in q for q in v["pq"])
        tag = " [arXiv_listing: cat-filter " + ("OK" if has_cat else "MISSING") + "]"
    print(f"  {k}: pq={len(v['pq'])} opt={len(v['opt'])}{tag}")
print(f"mandatory_total: {mand}")
print(f"optional_total: {opt}")
print(f"grand_total: {mand + opt}")
print(f"citation_chase_instructions: {chase}")
print(f"text_queries (mandatory - chase): {mand - chase}")
print(f"ieee_site_queries: {site}")
print(f"arxiv_cat_filter_queries: {cat}")

bare_tts = [q for v in lanes.values() for q in v["pq"] + v["opt"]
            if re.search(r"\bTTS\b", q)]
print(f"bare_TTS_violations: {len(bare_tts)}")
for q in bare_tts:
    print(f"  VIOLATION: {q}")

ok = (mand == 102 and opt == 3 and site == 3 and cat == 16 and len(bare_tts) == 0
      and len(lanes) == 21)
print("OVERALL:", "PASS" if ok else "FAIL (expected mand=102 opt=3 site=3 cat=16 tts=0 lanes=21)")
sys.exit(0 if ok else 1)
